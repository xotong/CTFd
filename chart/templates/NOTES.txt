CTFd has been deployed successfully!

1. Get the application URL by running:
{{- if and .Values.global.openshift.enabled .Values.global.openshift.route.enabled }}

  export ROUTE_HOST=$(oc get route {{ include "ctfd.fullname" . }} -o jsonpath='{.spec.host}')
  echo "CTFd URL: https://$ROUTE_HOST"

{{- else if .Values.nginx.enabled }}

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "ctfd.nginx.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo "CTFd URL: http://$SERVICE_IP"

{{- else }}

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "ctfd.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo "CTFd URL: http://$SERVICE_IP:{{ .Values.ctfd.service.port }}"

  Or use port-forward:
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "ctfd.fullname" . }} 8000:{{ .Values.ctfd.service.port }}
  echo "CTFd URL: http://localhost:8000"

{{- end }}
